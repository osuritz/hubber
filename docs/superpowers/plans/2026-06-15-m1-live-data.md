```swift
// Sources/Hubber/Core/SettingsStore.swift
import Foundation

@MainActor
final class SettingsStore: ObservableObject {
    private static let managedDomain = "com.hubber.app" as CFString

    private let defaults: UserDefaults

    // MARK: - SLO

    @Published var sloEnabled: Bool {
        didSet { defaults.set(sloEnabled, forKey: "reviewSLOEnabled") }
    }

    @Published var sloHours: Double {
        didSet { defaults.set(sloHours, forKey: "sloWindowHours") }
    }

    @Published var sloDisplay: SLODisplay {
        didSet { defaults.set(sloDisplay.rawValue, forKey: "sloDisplay") }
    }

    // MARK: - Incoming

    @Published var incomingMode: IncomingReviewMode {
        didSet { defaults.set(incomingMode.rawValue, forKey: "incomingMode") }
    }

    // MARK: - Outgoing

    @Published var unresolvedMode: UnresolvedThreadMode {
        didSet { defaults.set(unresolvedMode.rawValue, forKey: "unresolvedThreadMode") }
    }

    // MARK: - Appearance

    @Published var rowDensity: RowDensity {
        didSet { defaults.set(rowDensity.rawValue, forKey: "rowDensity") }
    }

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults

        sloEnabled = Self.bool(key: "reviewSLOEnabled", defaults: defaults, fallback: true)
        sloHours = Self.double(key: "sloWindowHours", defaults: defaults, fallback: 4.0)
        sloDisplay = SLODisplay(
            rawValue: Self.string(key: "sloDisplay", defaults: defaults, fallback: SLODisplay.remaining.rawValue)
        ) ?? .remaining
        incomingMode = IncomingReviewMode(
            rawValue: Self.string(key: "incomingMode", defaults: defaults, fallback: IncomingReviewMode.explicit.rawValue)
        ) ?? .explicit
        unresolvedMode = UnresolvedThreadMode(
            rawValue: Self.string(key: "unresolvedThreadMode", defaults: defaults, fallback: UnresolvedThreadMode.allOpen.rawValue)
        ) ?? .allOpen
        rowDensity = RowDensity(
            rawValue: Self.int(key: "rowDensity", defaults: defaults, fallback: RowDensity.normal.rawValue)
        ) ?? .normal
    }

    /// Returns true when a key is locked via MDM managed preferences.
    func isLocked(_ key: String) -> Bool {
        CFPreferencesIsAppKeyForced(key as CFString, Self.managedDomain)
    }

    /// SLO window in hours for a given repo. Per-repo overrides slot in here (backlog).
    func sloHours(for repo: String) -> Double {
        sloHours
    }

    // MARK: - MDM → UserDefaults → default priority

    private static func string(key: String, defaults: UserDefaults, fallback: String) -> String {
        if let managed = CFPreferencesCopyAppValue(key as CFString, managedDomain) as? String {
            return managed
        }
        return defaults.string(forKey: key) ?? fallback
    }

    private static func bool(key: String, defaults: UserDefaults, fallback: Bool) -> Bool {
        if CFPreferencesIsAppKeyForced(key as CFString, managedDomain),
           let managed = CFPreferencesCopyAppValue(key as CFString, managedDomain) as? Bool
        {
            return managed
        }
        guard defaults.object(forKey: key) != nil else { return fallback }
        return defaults.bool(forKey: key)
    }

    private static func double(key: String, defaults: UserDefaults, fallback: Double) -> Double {
        if CFPreferencesIsAppKeyForced(key as CFString, managedDomain),
           let managed = CFPreferencesCopyAppValue(key as CFString, managedDomain) as? Double
        {
            return managed
        }
        guard defaults.object(forKey: key) != nil else { return fallback }
        return defaults.double(forKey: key)
    }

    private static func int(key: String, defaults: UserDefaults, fallback: Int) -> Int {
        if CFPreferencesIsAppKeyForced(key as CFString, managedDomain),
           let managed = CFPreferencesCopyAppValue(key as CFString, managedDomain) as? Int
        {
            return managed
        }
        guard defaults.object(forKey: key) != nil else { return fallback }
        return defaults.integer(forKey: key)
    }
}
```

```swift
// Sources/Hubber/GitHub/GitHubProvider.swift
import Foundation
import SwiftUI

final class GitHubProvider: CodeHostProvider {
    private let session: URLSession

    init(session: URLSession = .shared) {
        self.session = session
    }

    // MARK: - Incoming Reviews

    func fetchIncomingReviews(
        accessToken: String,
        viewerLogin: String,
        mode: IncomingReviewMode
    ) async throws -> [IncomingReview] {
        var searchQuery = "is:pr is:open review-requested:@me"
        if mode == .team {
            searchQuery += " team-review-requested:@me"
        }

        let payload = try await graphQL(
            query: Self.incomingReviewsQuery,
            variables: ["searchQuery": searchQuery, "first": 100],
            accessToken: accessToken
        )

        guard let search = payload["search"] as? [String: Any],
              let nodes = search["nodes"] as? [[String: Any]]
        else { throw GitHubAuthError.invalidResponse }

        return nodes.compactMap { node in
            Self.parseIncomingReview(node: node, viewerLogin: viewerLogin)
        }
    }

    private static func parseIncomingReview(node: [String: Any], viewerLogin: String) -> IncomingReview? {
        guard
            node["__typename"] as? String == "PullRequest",
            node["isDraft"] as? Bool == false,
            let id = node["id"] as? String,
            let number = node["number"] as? Int,
            let title = node["title"] as? String,
            let repo = (node["repository"] as? [String: Any])?["nameWithOwner"] as? String,
            let authorNode = node["author"] as? [String: Any],
            let authorLogin = authorNode["login"] as? String
        else { return nil }

        let url = (node["url"] as? String).flatMap(URL.init(string:))
        let avatarURL = (authorNode["avatarUrl"] as? String).flatMap(URL.init(string:))
        let additions = node["additions"] as? Int ?? 0
        let deletions = node["deletions"] as? Int ?? 0
        let files = node["changedFiles"] as? Int ?? 0

        let sloStart = parseSLOClockStart(
            timelineNodes: (node["timelineItems"] as? [String: Any])?["nodes"] as? [[String: Any]] ?? [],
            viewerLogin: viewerLogin
        )

        let author = GitHubUser(
            login: authorLogin,
            initials: String(authorLogin.prefix(2)).uppercased(),
            color: colorForLogin(authorLogin),
            avatarURL: avatarURL
        )

        return IncomingReview(
            id: id,
            repo: repo,
            number: number,
            title: title,
            url: url,
            author: author,
            sloClockStart: sloStart,
            additions: additions,
            deletions: deletions,
            files: files
        )
    }

    /// SLO clock = max(earliest ReviewRequestedEvent for viewer, latest ReadyForReviewEvent).
    private static func parseSLOClockStart(
        timelineNodes: [[String: Any]],
        viewerLogin: String
    ) -> Date {
        let iso = ISO8601DateFormatter()
        var reviewRequestedAt: Date?
        var readyForReviewAt: Date?

        for item in timelineNodes {
            switch item["__typename"] as? String {
            case "ReviewRequestedEvent":
                guard let dateStr = item["createdAt"] as? String,
                      let date = iso.date(from: dateStr)
                else { continue }
                let reviewer = item["requestedReviewer"] as? [String: Any]
                let login = reviewer?["login"] as? String
                // Accept explicit user request or team request (no login field on Team).
                if login == viewerLogin || login == nil {
                    if reviewRequestedAt == nil || date < reviewRequestedAt! {
                        reviewRequestedAt = date
                    }
                }
            case "ReadyForReviewEvent":
                guard let dateStr = item["createdAt"] as? String,
                      let date = iso.date(from: dateStr)
                else { continue }
                if readyForReviewAt == nil || date > readyForReviewAt! {
                    readyForReviewAt = date
                }
            default:
                break
            }
        }

        let base = reviewRequestedAt ?? Date()
        if let ready = readyForReviewAt {
            return max(base, ready)
        }
        return base
    }

    // MARK: - GraphQL transport

    func graphQL(query: String, variables: [String: Any], accessToken: String) async throws -> [String: Any] {
        var request = URLRequest(url: URL(string: "https://api.github.com/graphql")!)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.setValue("application/vnd.github+json", forHTTPHeaderField: "Accept")
        request.setValue("Bearer \(accessToken)", forHTTPHeaderField: "Authorization")
        request.httpBody = try JSONSerialization.data(withJSONObject: ["query": query, "variables": variables])

        let (data, response) = try await session.data(for: request)
        guard let http = response as? HTTPURLResponse else { throw GitHubAuthError.invalidResponse }
        if http.statusCode == 401 { throw GitHubAuthError.badCredentials }
        guard (200..<300).contains(http.statusCode) else {
            throw GitHubAuthError.github("GitHub returned HTTP \(http.statusCode).")
        }

        let json = try JSONSerialization.jsonObject(with: data) as? [String: Any] ?? [:]
        if let errors = json["errors"] as? [[String: Any]],
           let message = errors.first?["message"] as? String {
            throw GitHubAuthError.github(message)
        }
        guard let payload = json["data"] as? [String: Any] else { throw GitHubAuthError.invalidResponse }
        return payload
    }

    // MARK: - Queries

    private static let incomingReviewsQuery = """
    query IncomingReviews($searchQuery: String!, $first: Int!) {
      search(query: $searchQuery, type: ISSUE, first: $first) {
        nodes {
          __typename
          ... on PullRequest {
            id
            number
            title
            url
            isDraft
            additions
            deletions
            changedFiles
            repository { nameWithOwner }
            author {
              login
              ... on User { avatarUrl }
            }
            timelineItems(first: 50, itemTypes: [REVIEW_REQUESTED_EVENT, READY_FOR_REVIEW_EVENT]) {
              nodes {
                __typename
                ... on ReviewRequestedEvent {
                  createdAt
                  requestedReviewer {
                    __typename
                    ... on User { login }
                    ... on Team { slug }
                  }
                }
                ... on ReadyForReviewEvent { createdAt }
              }
            }
          }
        }
      }
    }
    """
}
```

```swift
// MARK: - Outgoing PRs

func fetchOutgoingPullRequests(
    accessToken: String,
    viewerLogin: String,
    unresolvedMode: UnresolvedThreadMode
) async throws -> [OutgoingReview] {
    let payload = try await graphQL(
        query: Self.outgoingPRsQuery,
        variables: ["first": 100],
        accessToken: accessToken
    )

    guard let viewer = payload["viewer"] as? [String: Any],
          let prs = viewer["pullRequests"] as? [String: Any],
          let nodes = prs["nodes"] as? [[String: Any]]
    else { throw GitHubAuthError.invalidResponse }

    return nodes.compactMap { Self.parseOutgoingPR($0, unresolvedMode: unresolvedMode) }
}

private static func parseOutgoingPR(_ node: [String: Any], unresolvedMode: UnresolvedThreadMode) -> OutgoingReview? {
    guard
        node["isDraft"] as? Bool == false,
        let id = node["id"] as? String,
        let number = node["number"] as? Int,
        let title = node["title"] as? String,
        let repo = (node["repository"] as? [String: Any])?["nameWithOwner"] as? String
    else { return nil }

    let url = (node["url"] as? String).flatMap(URL.init(string:))
    let updatedAt = (node["updatedAt"] as? String).flatMap(ISO8601DateFormatter().date(from:)) ?? Date()
    let mergeable = (node["mergeable"] as? String) == "MERGEABLE"

    let (ci, ciDetail) = parseCIStatus(node)
    let reviewers = parseReviewers(node)
    let approvals = reviewers.filter { $0.state == .approved }.count
    let unresolved = countUnresolved(node, mode: unresolvedMode)

    return OutgoingReview(
        id: id,
        repo: repo,
        number: number,
        title: title,
        url: url,
        ci: ci,
        ciDetail: ciDetail,
        unresolved: unresolved,
        approvals: approvals,
        requiredApprovals: 1,  // GitHub API doesn't expose branch-protection required count without admin scope; default 1
        reviewers: reviewers,
        updatedAgoHours: max(0, Date().timeIntervalSince(updatedAt) / 3600),
        mergeable: mergeable
    )
}

private static func parseCIStatus(_ node: [String: Any]) -> (PullRequestCI, String) {
    guard
        let commits = node["commits"] as? [String: Any],
        let commitNodes = commits["nodes"] as? [[String: Any]],
        let rollup = commitNodes.first?["commit"] as? [String: Any],
        let statusNode = rollup["statusCheckRollup"] as? [String: Any]
    else { return (.pending, "No checks") }

    let state = statusNode["state"] as? String ?? ""
    let contexts = statusNode["contexts"] as? [String: Any]
    let total = contexts?["totalCount"] as? Int ?? 0
    let contextNodes = contexts?["nodes"] as? [[String: Any]] ?? []

    let failing = contextNodes.filter {
        let conclusion = $0["conclusion"] as? String
        let checkState = $0["state"] as? String
        return conclusion == "FAILURE" || conclusion == "ERROR"
            || checkState == "FAILURE" || checkState == "ERROR"
    }.count
    let running = contextNodes.filter {
        let status = $0["status"] as? String
        let checkState = $0["state"] as? String
        return status == "IN_PROGRESS" || status == "QUEUED"
            || checkState == "PENDING"
    }.count

    switch state {
    case "SUCCESS":
        return (.passing, "\(total)/\(total) checks")
    case "FAILURE", "ERROR":
        let passing = total - failing
        return (.failing, "\(failing) failing · \(passing) passing")
    case "PENDING":
        if running > 0 {
            let done = total - running
            return (.running, "\(done)/\(total) checks · running")
        }
        return (.pending, "Pending")
    default:
        return (.pending, "No checks")
    }
}

private static func parseReviewers(_ node: [String: Any]) -> [Reviewer] {
    guard
        let reviews = node["latestReviews"] as? [String: Any],
        let reviewNodes = reviews["nodes"] as? [[String: Any]]
    else { return [] }

    return reviewNodes.compactMap { review -> Reviewer? in
        guard
            let author = review["author"] as? [String: Any],
            let login = author["login"] as? String,
            let stateStr = review["state"] as? String
        else { return nil }

        let state: ReviewerState
        switch stateStr {
        case "APPROVED": state = .approved
        case "CHANGES_REQUESTED": state = .changesRequested
        case "COMMENTED", "DISMISSED": state = .commented
        default: state = .pending
        }

        let avatarURL = (author["avatarUrl"] as? String).flatMap(URL.init(string:))
        let user = GitHubUser(
            login: login,
            initials: String(login.prefix(2)).uppercased(),
            color: colorForLogin(login),
            avatarURL: avatarURL
        )

        return Reviewer(user: user, state: state)
    }
}

private static func countUnresolved(_ node: [String: Any], mode: UnresolvedThreadMode) -> Int {
    guard
        let threads = node["reviewThreads"] as? [String: Any],
        let threadNodes = threads["nodes"] as? [[String: Any]]
    else { return 0 }

    return threadNodes.filter { thread in
        guard
            thread["isResolved"] as? Bool == false,
            thread["isOutdated"] as? Bool == false
        else { return false }

        switch mode {
        case .allOpen:
            return true
        case .reviewOnly:
            let comment = (thread["comments"] as? [String: Any])?["nodes"] as? [[String: Any]]
            return comment?.first?["pullRequestReview"] != nil
        case .changesRequested:
            let comment = (thread["comments"] as? [String: Any])?["nodes"] as? [[String: Any]]
            let review = comment?.first?["pullRequestReview"] as? [String: Any]
            return review?["state"] as? String == "CHANGES_REQUESTED"
        }
    }.count
}
```

## Task 10: Avatar Fetching with Cache

- [ ] **Step 1: Create `AvatarCache`**

```swift
// Sources/Hubber/GitHub/AvatarCache.swift
import AppKit
import Foundation

@MainActor
final class AvatarCache {
    static let shared = AvatarCache()

    private let cache = NSCache<NSURL, NSImage>()
    private var inFlight: [URL: Task<NSImage?, Never>] = [:]

    func image(for url: URL) async -> NSImage? {
        if let cached = cache.object(forKey: url as NSURL) {
            return cached
        }
        if let existing = inFlight[url] {
            return await existing.value
        }
        let task = Task<NSImage?, Never> {
            guard let (data, _) = try? await URLSession.shared.data(from: url),
                  let image = NSImage(data: data)
            else { return nil }
            cache.setObject(image, forKey: url as NSURL)
            return image
        }
        inFlight[url] = task
        let result = await task.value
        inFlight.removeValue(forKey: url)
        return result
    }
}
```
